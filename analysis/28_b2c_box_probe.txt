###############################################################################
# Caddyfile — Authelia forward_auth in front of trading_corp.
###############################################################################
# Two vhosts:
#
#   trading.jacksumner.com — the dashboard. Most paths are gated through
#       Authelia (forward_auth). The bypass list at the top covers paths
#       that genuinely shouldn't require interactive auth: TradingView's
#       webhook (HMAC-authenticated separately), the healthz endpoint,
#       and PWA bootstrap files (service worker, manifest, icons) that
#       must be fetchable WITHOUT a session cookie for iOS Add-to-Home-
#       Screen install to succeed.
#
#   auth.jacksumner.com — Authelia's own login UI. Always public (you
#       can't log in if the login page is gated by login!).
#
# Both auto-issue Let's Encrypt certs on first request to the new host.
###############################################################################

# ── trading.jacksumner.com — protected dashboard ──────────────────────────
trading.jacksumner.com {
    # Bypass list: paths that skip Authelia. Order matters — public matchers
    # are evaluated first; anything not matched falls through to the
    # forward_auth-protected handle below.
    @public {
        # TradingView webhook — HMAC + IP allowlist + replay protection live
        # inside trading_corp. Authelia would block TV's POSTs since they
        # don't carry our session cookie.
        path /webhook/tradingview/lord-otter
        path /webhook/tradingview/market-cypher

        # Liveness probe. Health checks shouldn't depend on the auth backend
        # — if Authelia is down we want trading_corp's own health visible.
        path /healthz

        # PWA bootstrap. iOS Add-to-Home-Screen fetches these WITHOUT
        # session cookies during install. If they're gated, install breaks.
        path /sw.js
        path /manifest.webmanifest
        path /offline.html
        path /static/*
    }

    handle @public {
        reverse_proxy localhost:8000
    }

    # Default handle — everything else goes through Authelia first.
    # forward_auth: Caddy makes a sub-request to Authelia for every
    # incoming request. If Authelia returns 200, Caddy proxies to
    # trading_corp. If 401/302, Caddy follows Authelia's redirect to
    # the login page (auth.jacksumner.com).
    handle {
        forward_auth localhost:9091 {
            uri /api/authz/forward-auth

            # Headers Authelia sets on success — pass them through to
            # trading_corp so the dashboard could (later) read 'who am I'.
            # Currently unused on the trading_corp side but harmless.
            copy_headers Remote-User Remote-Groups Remote-Name Remote-Email
        }
        reverse_proxy localhost:8000
    }

    log {
        output stderr
        format console
    }
}

# ── auth.jacksumner.com — Authelia login UI ───────────────────────────────
auth.jacksumner.com {
    reverse_proxy localhost:9091

    log {
        output stderr
        format console
    }
}

# predictions.jacksumner.com -> pm_web (loopback :8081)
predictions.jacksumner.com {

}
reverse_proxy localhost:8081
}

}
}
